cmake_minimum_required(VERSION 3.8)
project(my_arm_with_ball_moveit_config)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(geometric_shapes REQUIRED)
find_package(ament_index_cpp REQUIRED)
add_executable(planning_scene_loader src/planning_scene_loader.cpp)
ament_target_dependencies(planning_scene_loader rclcpp moveit_msgs geometry_msgs shape_msgs geometric_shapes ament_index_cpp)
install(TARGETS planning_scene_loader DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/validate_planning.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch meshes DESTINATION share/${PROJECT_NAME})
install(FILES .setup_assistant DESTINATION share/${PROJECT_NAME})
ament_package()
